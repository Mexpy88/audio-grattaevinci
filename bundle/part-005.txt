ow new Error(`Per ${line.article}${line.size?' · '+line.size:''} inserisci sia i cartoni presi sia i pezzi reali da scaricare.`);
      const rem=this.deliveredForLine(r,line.id),already=allocatedCartons.get(line.id)||0,remainingCartons=Math.max(0,line.cartonsRequested-rem.cartons-already);if(cartons>remainingCartons)throw new Error(`Troppi cartoni per ${line.article}: residui ${remainingCartons}.`);allocatedCartons.set(line.id,already+cartons);items.push({lineId:line.id,item:source,source,stockRow:source,pieces,cartons,extra:false})}
    if(!items.length)throw new Error('Inserisci almeno un prelievo con cartoni e pezzi.');
    const issue=this.issues.create({destination:r.destination,note,requestId:r.id,items});r.deliveries.push({id:uid(),issueId:issue.id,createdAt:issue.createdAt,operator:this.auth.user,note,items:clone(issue.items)});const s=this.summary(r),before=r.status;r.status=s.remaining===0?'COMPLETATA':s.delivered>0?'PARZIALE':'DA PREPARARE';if(r.status==='COMPLETATA')r.completedAt=nowIso();else r.completedAt='';this.notifications.create({recipient:r.operator||'Lina',title:`Richiesta ${r.status.toLowerCase()} · ${r.id}`,body:`${s.delivered}/${s.requested} cartoni · ${s.pieces} pezzi scaricati`,type:'REQUEST_PROGRESS',entityType:'REQUEST',entityId:r.id});this.audit.record('REQUEST_DELIVERY','REQUEST',r.id,{status:before},{status:r.status,issueId:issue.id,summary:s});this.store.save('request:fulfill');return issue
  }
}

class MovementService{
  constructor(store,stock,ledger,issues,receiving,audit,auth){Object.assign(this,{store,stock,ledger,issues,receiving,audit,auth})}
  issue({sourceKey,quantity,destination,note}){if(!this.auth.can('MOVE'))throw new Error('Operazione non autorizzata.');const source=this.stock.findByKey(sourceKey);if(!source)throw new Error('Giacenza non trovata.');const issue=this.issues.create({destination,note,items:[{item:source,source,stockRow:source,pieces:integer(quantity)}]});this.store.save('movement:issue');return issue}
  transfer({sourceKey,quantity,location,pallet,note}){if(!this.auth.can('MOVE'))throw new Error('Operazione non autorizzata.');const source=this.stock.findByKey(sourceKey);if(!source)throw new Error('Giacenza non trovata.');const target=cleanBucket({location,pallet});if(!target.location)throw new Error('Inserisci la nuova Fila/Scaffale.');const e=this.ledger.transfer(source,source,target,integer(quantity),{kind:'MOVE',note:String(note||'')});if(source.location===CONFIG.receivingLocation){const found=this.receiving.consumeFromReceiving(source,integer(quantity),{reason:'SPOSTAMENTO'});if(found)e.source.receiptLineId=found.line.id}this.store.save('movement:transfer');return e}
  cancelLedger(id){const e=this.store.db.ledger.find(x=>x.id===id);if(!e)throw new Error('Movimento non trovato.');const out=this.ledger.cancel(id);if(e.source?.receiptLineId&&(e.type==='OUT'||e.type==='TRANSFER'))this.receiving.restoreToReceiving(e.source.receiptLineId,e.quantity);this.store.save('ledger:cancel');return out}
}

class RectificationService{
  constructor(store,stock,ledger,audit,auth){Object.assign(this,{store,stock,ledger,audit,auth})}
  setQuantity({currentKey='',article,size,state,location,pallet,counted,note=''}){if(!this.auth.can('RECTIFY'))throw new Error('Operazione non autorizzata.');const current=currentKey?this.stock.findByKey(currentKey):null,item=current?cleanItem(current):cleanItem({article,size,state}),bucket=current?cleanBucket(current):cleanBucket({location,pallet}),old=current?.quantity||0,target=integer(counted),delta=target-old;if(!item.article)throw new Error('Inserisci l’articolo.');if(!bucket.location)throw new Error('Inserisci Fila/Scaffale.');const e=this.ledger.adjust(item,bucket,delta,{kind:'RECTIFICATION',note:String(note||'')});this.audit.record('RECTIFICATION','STOCK',currentKey||bucketKey({...item,...bucket}),{quantity:old},{quantity:target},{note});this.store.save('stock:rectify');return e}
}

function activityForDay(db,key,user=''){
  const led=(db.ledger||[]).filter(e=>!e.cancelledAt&&dateKeyLocal(e.at)===key),receipts=(db.receipts||[]).filter(r=>dateKeyLocal(r.receivedAt||r.createdAt)===key),issues=(db.issues||[]).filter(i=>dateKeyLocal(i.createdAt)===key&&(i.movementIds||[]).some(id=>(db.ledger||[]).some(e=>e.id===id&&!e.cancelledAt))),transfers=led.filter(e=>e.type==='TRANSFER'&&e.source?.kind!=='PUTAWAY'),visibleReq=(db.requests||[]).filter(r=>user!=='Lina'||r.operator==='Lina'||r.destination==='LINA'),received=visibleReq.filter(r=>dateKeyLocal(r.requestedAt||r.createdAt)===key).length,completed=visibleReq.filter(r=>dateKeyLocal(r.completedAt)===key).length,audits=(db.audits||[]).filter(a=>dateKeyLocal(a.at)===key),latest=[...audits].sort((a,b)=>new Date(b.at)-new Date(a.at))[0]?.at||'';return{car:receipts.length,scar:issues.length,sp:transfers.length,received,completed,latest}
}
