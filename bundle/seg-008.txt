[]).reduce((n,x)=>n+number(x.quantity),0);return{id:l.id||uid(),article:normalizeArticle(l.article_base),size:norm(l.size),state:norm(l.state||'NUOVO'),ddtQuantity:number(l.ddt_quantity),checkedQuantity:checked,remainingToPutAway:Math.max(0,checked-putAway),putAwayQuantity:putAway,pallet:norm(l.pallet||'P01'),receivingPallet:norm(l.receiving_pallet||l.temp_bancale||l.pallet||'P01'),note:l.note||'',suggestedLocation:norm(l.suggested_location),suggestedPallet:norm(l.suggested_bancale),partials:[]}})}));
  out.audits=(legacy?.audits||[]).map(a=>({id:a.id||uid(),action:a.action||'',entityType:a.entityType||a.entity_type||'',entityId:a.entityId||a.entity_id||'',operator:a.operator||'',at:a.at||nowIso(),before:a.before||null,after:a.after||null,meta:{legacy:true}}));out.notifications=(legacy?.notifications||[]).map(n=>({id:n.id||uid(),recipient:n.recipient||'',title:n.title||'',body:n.body||'',type:n.type||'INFO',entityType:n.entityType||'',entityId:n.entityId||'',createdAt:n.createdAt||n.created_at||nowIso(),readAt:n.readAt||n.read_at||null}));
  out.counters={receipt:integer(legacy?.counters?.goods_receipt),request:integer(legacy?.counters?.request),issue:integer(legacy?.counters?.scarico)};out.settings.legacyMigratedAt=nowIso();return out
}
