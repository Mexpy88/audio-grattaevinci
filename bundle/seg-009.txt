async function ensureXlsx(){

  if(!xlsxPromise)xlsxPromise=new Promise((resolve,reject)=>{

function detectHeader(matrix){let best={row:0,score:-1,cols:{}};for(let r=0;r<Math.min(matrix.length,30);r++){const headers=matrix[r]||[],cols={};for(const [k,names] of Object.entries(aliases))cols[k]=findCol(headers,names);const score=(cols.article>=0?6:0)+(cols.size>=0?1:0)+(cols.location>=0||cols.fila>=0?2:0)+(cols.pallet>=0?1:0)+(cols.state>=0&&cols.quantity>=0?4:0)+CONFIG.states.filter(s=>cols[s]>=0).length;if(score>best.score)best={row:r,score,cols}}if(best.cols.article<0)throw new Error('Colonna CODICE SAP / ARTICOLO non trovata nel Master.');return best}
function readLocation(row,cols){if(cols.location>=0)return norm(row[cols.location]);return [cols.fila>=0?norm(row[cols.fila]):'',cols.scaffale>=0?norm(row[cols.scaffale]):''].filter(Boolean).join('/')}
function parseMaster(wb){const sheetName=wb.SheetNames.includes('MAGAZZINO')?'MAGAZZINO':wb.SheetNames.find(n=>!['APP_DATI','NOVA_DATI'].includes(n))||wb.SheetNames[0],ws=wb.Sheets[sheetName];if(!ws)throw new Error('Foglio Master non trovato.');const matrix=X().utils.sheet_to_json(ws,{header:1,defval:'',raw:true,blankrows:false}),det=detectHeader(matrix),mode=det.cols.state>=0&&det.cols.quantity>=0?'long':'wide',rows=[];
  for(let r=det.row+1;r<matrix.length;r++){const row=matrix[r]||[],article=normalizeArticle(row[det.cols.article]);if(!article)continue;const size=det.cols.size>=0?norm(row[det.cols.size]):'',location=readLocation(row,det.cols),pallet=det.cols.pallet>=0?norm(row[det.cols.pallet]):'',description=det.cols.description>=0?String(row[det.cols.description]||'').trim():'';
    if(mode==='long'){const state=norm(row[det.cols.state]),quantity=Number(row[det.cols.quantity]||0)||0;if(state&&quantity)rows.push({article,size,state,quantity,location,pallet,description,sourceRow:r})}
    else for(const state of CONFIG.states){const c=det.cols[state];if(c<0)continue;const quantity=Number(row[c]||0)||0;if(quantity)rows.push({article,size,state,quantity,location,pallet,description,sourceRow:r})}
  }
  return{rows:normalizeMasterRows(rows),sheetName,headerRow:det.row,columns:det.cols,mode,rowCount:Math.max(0,matrix.length-det.row-1)}
}
