const number=value=>Number(value||0)||0;
const integer=value=>Math.max(0,Math.floor(Number(value)||0));
const locationOf=x=>norm(x?.location||x?.fila_scaffale||[x?.fila,x?.scaffale].filter(Boolean).join('/'));
const palletOf=x=>norm(x?.pallet||x?.bancale);
const bucketKey=b=>[normalizeArticle(b.article||b.article_base),norm(b.size||b.taglia),norm(b.state),locationOf(b),palletOf(b)].join('|');
const cleanItem=item=>({article:normalizeArticle(item?.article||item?.article_base),size:norm(item?.size||item?.taglia),state:norm(item?.state||'NUOVO')||'NUOVO'});
const cleanBucket=b=>({location:locationOf(b),pallet:palletOf(b)});

function normalizeMasterRows(rows=[]){
  const out=[];
  for(const raw of rows){
    const article=normalizeArticle(raw.article||raw.article_base||raw.codice||raw.codice_sap),size=norm(raw.size||raw.taglia),location=locationOf(raw),pallet=palletOf(raw),description=String(raw.description||raw.descrizione||'').trim();
    if(!article)continue;
    if(raw.state&&raw.quantity!=null){out.push({id:raw.id||uid(),article,size,state:norm(raw.state),quantity:number(raw.quantity),location,pallet,description,sourceRow:raw.sourceRow??null});continue}
    for(const state of CONFIG.states){const q=number(raw[state]??raw[state.toLowerCase()]);if(q)out.push({id:uid(),article,size,state,quantity:q,location,pallet,description,sourceRow:raw.sourceRow??null})}
  }
  return out;
}

class StockService{
  constructor(store){this.store=store;this.cache=null;this.store?.bus?.on?.('db:changed',()=>this.invalidate())}
  invalidate(){this.cache=null}
  rows({excludeEventId=''}={}){
    if(!excludeEventId&&this.cache)return this.cache;
    const map=new Map(),apply=(bucket,delta)=>{const b={...cleanItem(bucket),...cleanBucket(bucket)},key=bucketKey(b);if(!map.has(key))map.set(key,{...b,quantity:0});map.get(key).quantity+=number(delta)};
    for(const r of normalizeMasterRows(this.store.db.master.rows))apply(r,r.quantity);
    for(const e of this.store.db.ledger){if(e.cancelledAt||e.id===excludeEventId||e.affectsStock===false)continue;const q=number(e.quantity);
      if(e.type==='IN')apply({...e.item,...e.to},q);
      else if(e.type==='OUT')apply({...e.item,...e.from},-q);
      else if(e.type==='TRANSFER'){apply({...e.item,...e.from},-q);apply({...e.item,...e.to},q)}
      else if(e.type==='ADJUST')apply({...e.item,...e.atBucket},number(e.delta));
    }
    const rows=[...map.values()].filter(r=>Math.abs(r.quantity)>1e-9).sort((a,b)=>a.article.localeCompare(b.article,'it')||a.size.localeCompare(b.size,'it')||a.location.localeCompare(b.location,'it')||a.pallet.localeCompare(b.pallet,'it'));
    if(!excludeEventId)this.cache=rows;
    return rows;
  }
  positive(opts){return this.rows(opts).filter(r=>r.quantity>0)}
  findByKey(key,opts){return this.rows(opts).find(r=>bucketKey(r)===key)}
  sources(article,size='',state=''){const A=normalizeArticle(article),S=norm(size),ST=norm(state);return this.positive().filter(r=>r.article===A&&(!S||r.size===S)&&(!ST||r.state===ST)).sort((a,b)=>Number(a.location===CONFIG.receivingLocation)-Number(b.location===CONFIG.receivingLocation)||b.quantity-a.quantity)}
  total(article,size='',state=''){return this.sources(article,size,state).reduce((n,r)=>n+r.quantity,0)}
  receivingTotal(article,size=''){return this.sources(article,size).filter(r=>r.location===CONFIG.receivingLocation).reduce((n,r)=>n+r.quantity,0)}
  suggest(article,size='',state=''){return this.sources(article,size,state).filter(r=>r.location!==CONFIG.receivingLocation).sort((a,b)=>b.quantity-a.quantity)[0]||null}
  assertAvailable(sourceKey,amount,{excludeEventId=''}={}){const r=this.findByKey(sourceKey,{excludeEventId}),q=number(amount);if(!r||r.quantity+1e-9<q)throw new Error(`Disponibilità insufficiente: ${r?.quantity||0} disponibili, ${q} richiesti.`);return r}
}
