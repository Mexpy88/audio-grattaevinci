class IssueService{
  constructor(store,stock,ledger,audit,receiving,auth){Object.assign(this,{store,stock,ledger,audit,receiving,auth})}
  nextId(){this.store.db.counters.issue=(this.store.db.counters.issue||0)+1;return `SC-${new Date().getFullYear()}-${String(this.store.db.counters.issue).padStart(5,'0')}`}
  prepare(items=[]){
    const prepared=[];
    for(const p of items){
      const source=p.source||p,stockRow=p.stockRow||source,item=cleanItem(p.item||stockRow),pieces=integer(p.pieces??p.quantity),cartons=integer(p.cartons);
      if(pieces<=0)continue;
      const bucket=cleanBucket(source),sourceKey=bucketKey({...item,...bucket});
      prepared.push({raw:p,item,source:bucket,stockRow:{...stockRow,...item,...bucket},sourceKey,pieces,cartons,lineId:p.lineId||'',extra:!!p.extra});
    }
    if(!prepared.length)throw new Error('Nessuna quantità da scaricare.');
    const totals=new Map();for(const p of prepared)totals.set(p.sourceKey,(totals.get(p.sourceKey)||0)+p.pieces);
    for(const [key,q] of totals)this.stock.assertAvailable(key,q);
    const receivingTotals=new Map();for(const p of prepared)if(p.source.location===CONFIG.receivingLocation)receivingTotals.set(p.sourceKey,(receivingTotals.get(p.sourceKey)||0)+p.pieces);
    for(const [key,q] of receivingTotals){const row=this.stock.findByKey(key),found=row?this.receiving.findLineByPallet(row.pallet,row):null;if(!found)throw new Error('Pallet in Area Ricevimento non associato a un’entrata merci.');if(number(found.line.remainingToPutAway)+1e-9<q)throw new Error(`Sul pallet risultano ${number(found.line.remainingToPutAway)} pezzi ancora gestibili, ma ne stai prelevando ${q}.`)}
    return prepared;
  }
  create({destination='',note='',requestId='',items=[]}){
    if(!this.auth.can('MOVE')&&!this.auth.can('REQUEST_PROCESS'))throw new Error('Operazione non autorizzata.');
    const prepared=this.prepare(items),id=this.nextId(),issue={id,createdAt:nowIso(),operator:this.auth.user,destination:norm(destination),note:String(note||'').trim(),requestId,movementIds:[],items:[]};
    for(const p of prepared){const evt=this.ledger.outgoing(p.item,p.source,p.pieces,{kind:requestId?'REQUEST':'ISSUE',issueId:id,requestId,destination:issue.destination,note:issue.note});if(p.source.location===CONFIG.receivingLocation){const found=this.receiving.consumeFromReceiving(p.stockRow,p.pieces,{reason:requestId?'RICHIESTA':'SCARICO'});if(found)evt.source.receiptLineId=found.line.id}issue.movementIds.push(evt.id);issue.items.push({article:p.item.article,size:p.item.size,state:p.item.state,pieces:p.pieces,cartons:p.cartons,source:p.source,movementId:evt.id,lineId:p.lineId,extra:p.extra})}
    this.store.db.issues.unshift(issue);this.audit.record('ISSUE_CONFIRMED','ISSUE',id,null,issue);return issue
  }
}
